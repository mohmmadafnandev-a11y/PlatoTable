## 📁 **Database Schema**

### 1. `restaurants` Table
```sql
CREATE TABLE restaurants (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  subdomain TEXT UNIQUE NOT NULL,
  owner_email TEXT NOT NULL,
  logo_url TEXT,
  phone TEXT,
  address TEXT,
  currency TEXT DEFAULT 'PKR',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

### 2. `restaurant_admins` Table
```sql
CREATE TABLE restaurant_admins (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  user_id UUID REFERENCES auth.users(id) ON DELETE CASCADE,
  username TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  role TEXT DEFAULT 'admin',
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

### 3. `categories` Table
```sql
CREATE TABLE categories (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  name TEXT NOT NULL,
  icon TEXT,
  display_order INTEGER DEFAULT 0,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

### 4. `menu_items` Table
```sql
CREATE TABLE menu_items (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  category_id UUID REFERENCES categories(id) ON DELETE SET NULL,
  name TEXT NOT NULL,
  description TEXT,
  price DECIMAL(10,2) NOT NULL,
  image_url TEXT,
  is_available BOOLEAN DEFAULT TRUE,
  is_offer BOOLEAN DEFAULT FALSE,
  offer_price DECIMAL(10,2),
  offer_description TEXT,
  offer_start_date TIMESTAMP,
  offer_end_date TIMESTAMP,
  display_order INTEGER DEFAULT 0,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

### 5. `qr_tokens` Table
```sql
CREATE TABLE qr_tokens (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  table_number TEXT NOT NULL,
  token TEXT UNIQUE NOT NULL,
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP DEFAULT NOW() + INTERVAL '15 minutes'
);

CREATE INDEX idx_qr_tokens_token ON qr_tokens(token);
CREATE INDEX idx_qr_tokens_restaurant ON qr_tokens(restaurant_id);
CREATE INDEX idx_qr_tokens_expires ON qr_tokens(expires_at);
```

### 6. `scan_logs` Table
```sql
CREATE TABLE scan_logs (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  table_number TEXT NOT NULL,
  token TEXT NOT NULL,
  ip_address TEXT,
  user_agent TEXT,
  device_type TEXT,
  scanned_at TIMESTAMP DEFAULT NOW(),
  expires_at TIMESTAMP DEFAULT NOW() + INTERVAL '15 minutes'
);

CREATE INDEX idx_scan_logs_restaurant ON scan_logs(restaurant_id);
CREATE INDEX idx_scan_logs_scanned ON scan_logs(scanned_at);
```

### 7. `settings` Table
```sql
CREATE TABLE settings (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  restaurant_id UUID REFERENCES restaurants(id) ON DELETE CASCADE,
  admin_username TEXT DEFAULT 'admin',
  admin_password_hash TEXT,
  qr_expiry_minutes INTEGER DEFAULT 15,
  menu_display_style TEXT DEFAULT 'grid',
  language TEXT DEFAULT 'en',
  theme_color TEXT DEFAULT '#4F46E5',
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);
```

## 🔐 **Row Level Security (RLS) Policies**

```sql
-- Enable RLS on all tables
ALTER TABLE restaurants ENABLE ROW LEVEL SECURITY;
ALTER TABLE restaurant_admins ENABLE ROW LEVEL SECURITY;
ALTER TABLE categories ENABLE ROW LEVEL SECURITY;
ALTER TABLE menu_items ENABLE ROW LEVEL SECURITY;
ALTER TABLE qr_tokens ENABLE ROW LEVEL SECURITY;
ALTER TABLE scan_logs ENABLE ROW LEVEL SECURITY;
ALTER TABLE settings ENABLE ROW LEVEL SECURITY;

-- Policies for restaurants table
CREATE POLICY "Admins can view their restaurant" ON restaurants
  FOR SELECT USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = restaurants.id
    )
  );

CREATE POLICY "Admins can update their restaurant" ON restaurants
  FOR UPDATE USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = restaurants.id
    )
  );

-- Policies for categories table
CREATE POLICY "Anyone can view categories" ON categories
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage categories" ON categories
  FOR ALL USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = categories.restaurant_id
    )
  );

-- Policies for menu_items table
CREATE POLICY "Anyone can view menu items" ON menu_items
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage menu items" ON menu_items
  FOR ALL USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = menu_items.restaurant_id
    )
  );

-- Policies for qr_tokens table
CREATE POLICY "Anyone can view qr tokens" ON qr_tokens
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage qr tokens" ON qr_tokens
  FOR ALL USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = qr_tokens.restaurant_id
    )
  );

-- Policies for scan_logs table
CREATE POLICY "Anyone can view scan logs" ON scan_logs
  FOR SELECT USING (true);

CREATE POLICY "Admins can manage scan logs" ON scan_logs
  FOR ALL USING (
    auth.uid() IN (
      SELECT user_id FROM restaurant_admins 
      WHERE restaurant_id = scan_logs.restaurant_id
    )
  );
```

# 🔷 PHASE 4: USER APP - CORE FEATURES (Weeks 7-8)

## 📱 **Scan Page** (`/scan/:token`)

### Token Validation (Entry point)
1. Extract token from URL
2. Query Supabase for token validity
3. Check if token exists
4. Check if token is active (`is_active = true`)
5. Check if token is expired (`expires_at > NOW()`)
6. Log the scan to `scan_logs`
7. Redirect to error page or menu
